```python
# 幂等写入：唯一索引兜底
import sqlite3
con = sqlite3.connect(":memory:")
con.execute("CREATE TABLE ev (k TEXT UNIQUE, v TEXT)")
def put(k, v):
    con.execute("INSERT OR IGNORE INTO ev VALUES (?, ?)", (k, v))
put("o-1", "paid"); put("o-1", "paid")
print(con.execute("SELECT count(*) FROM ev").fetchone())
```
